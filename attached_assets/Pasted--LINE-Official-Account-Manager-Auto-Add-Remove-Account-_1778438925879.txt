"""
LINE Official Account Manager - Auto Add & Remove Account
==========================================================
ฟังก์ชัน:
  1. เปิดหน้า ตั้งค่ากลุ่ม (ใช้ Chrome Profile ที่ login ไว้แล้ว)
  2. กดปุ่ม "เพิ่ม" → เลือกบัญชีใหม่ → กด "เพิ่มบัญชี"
  3. ลบบัญชีเก่าออกจากกลุ่ม

Requirements:
  pip install selenium webdriver-manager
"""

import os
import time
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.webdriver.chrome.service import Service
from selenium.common.exceptions import TimeoutException, NoSuchElementException
from webdriver_manager.chrome import ChromeDriverManager

# ============================================================
# ⚙️  ตั้งค่าตรงนี้
# ============================================================
GROUP_SETTINGS_URL = "https://manager.line.biz/account/YOUR_ACCOUNT_ID/group/YOUR_GROUP_ID/setting"

NEW_ACCOUNT_NAME   = "Vv72-FT"        # ชื่อบัญชีที่ต้องการ เพิ่ม
OLD_ACCOUNT_NAME   = "BT678🍓ฝาก"    # ชื่อบัญชีที่ต้องการ ลบ

WAIT_TIMEOUT       = 15               # วินาที รอ element โหลด

# --- Chrome Profile ที่ login LINE OA ไว้แล้ว ---
# Windows: C:\Users\<ชื่อ user>\AppData\Local\Google\Chrome\User Data
# macOS  : /Users/<ชื่อ user>/Library/Application Support/Google/Chrome
# Linux  : /home/<ชื่อ user>/.config/google-chrome
import platform
if platform.system() == "Windows":
    CHROME_PROFILE_DIR = os.path.join(os.environ["LOCALAPPDATA"], "Google", "Chrome", "User Data")
elif platform.system() == "Darwin":
    CHROME_PROFILE_DIR = os.path.expanduser("~/Library/Application Support/Google/Chrome")
else:
    CHROME_PROFILE_DIR = os.path.expanduser("~/.config/google-chrome")

CHROME_PROFILE_NAME = "Default"   # เปลี่ยนเป็น "Profile 1", "Profile 2" ฯลฯ ถ้าใช้หลาย profile
# ============================================================


def init_driver() -> webdriver.Chrome:
    """เปิด Chrome โดยใช้ Profile ที่ login ไว้แล้ว ไม่ต้อง login ซ้ำ"""
    options = webdriver.ChromeOptions()
    options.add_argument(f"--user-data-dir={CHROME_PROFILE_DIR}")
    options.add_argument(f"--profile-directory={CHROME_PROFILE_NAME}")
    options.add_argument("--start-maximized")
    options.add_argument("--disable-notifications")
    options.add_experimental_option("excludeSwitches", ["enable-automation"])
    options.add_experimental_option("useAutomationExtension", False)
    driver = webdriver.Chrome(
        service=Service(ChromeDriverManager().install()),
        options=options
    )
    # ซ่อน navigator.webdriver เพื่อป้องกัน bot detection
    driver.execute_cdp_cmd("Page.addScriptToEvaluateOnNewDocument", {
        "source": "Object.defineProperty(navigator, 'webdriver', {get: () => undefined})"
    })
    return driver


def wait_click(driver, by, selector, timeout=WAIT_TIMEOUT):
    """รอให้ element คลิกได้แล้วคลิก"""
    el = WebDriverWait(driver, timeout).until(
        EC.element_to_be_clickable((by, selector))
    )
    el.click()
    return el


def wait_visible(driver, by, selector, timeout=WAIT_TIMEOUT):
    """รอให้ element มองเห็น"""
    return WebDriverWait(driver, timeout).until(
        EC.visibility_of_element_located((by, selector))
    )


# ----------------------------------------------------------
# STEP 1 : กดปุ่ม "เพิ่ม" และเลือกบัญชีใหม่
# ----------------------------------------------------------
def add_new_account(driver: webdriver.Chrome, account_name: str):
    print(f"\n[ADD] กำลังเพิ่มบัญชี: {account_name}")

    # กดปุ่ม "เพิ่ม" (ปุ่มสีเขียวในส่วน การจัดการบัญชี)
    add_btn = wait_click(
        driver,
        By.XPATH,
        "//button[contains(@class,'btn') and normalize-space()='เพิ่ม']"
    )
    print("  ✔ กดปุ่ม 'เพิ่ม' แล้ว")
    time.sleep(1.5)

    # รอ modal เปิด
    wait_visible(driver, By.XPATH, "//div[contains(@class,'modal') and .//*[contains(text(),'เพิ่มบัญชีนี้ที่กลุ่ม')]]")
    print("  ✔ Modal เปิดแล้ว")

    # ---------- เลื่อนหาชื่อบัญชีใน list ฝั่งซ้าย ----------
    account_found = False
    max_scroll = 10

    for _ in range(max_scroll):
        # หา item ที่ตรงชื่อ
        items = driver.find_elements(
            By.XPATH,
            f"//div[contains(@class,'modal')]//li[normalize-space()='{account_name}'] | "
            f"//div[contains(@class,'modal')]//div[contains(@class,'account-item') and contains(normalize-space(),'{account_name}')]"
        )
        if items:
            items[0].click()
            print(f"  ✔ เลือกบัญชี '{account_name}' แล้ว")
            account_found = True
            break

        # scroll ลงใน list
        scrollable = driver.find_element(
            By.XPATH,
            "//div[contains(@class,'modal')]//ul[contains(@class,'account-list')] | "
            "//div[contains(@class,'modal')]//div[contains(@class,'list-wrap')]"
        )
        driver.execute_script("arguments[0].scrollTop += 150", scrollable)
        time.sleep(0.5)

    if not account_found:
        raise ValueError(f"ไม่พบบัญชี '{account_name}' ใน list กรุณาตรวจสอบชื่อ")

    time.sleep(0.5)

    # กดปุ่ม "เพิ่มบัญชี" (สีเขียว ใน modal)
    wait_click(
        driver,
        By.XPATH,
        "//div[contains(@class,'modal')]//button[contains(normalize-space(),'เพิ่มบัญชี')]"
    )
    print("  ✔ กดปุ่ม 'เพิ่มบัญชี' แล้ว")
    time.sleep(2)

    # รอ modal ปิด
    WebDriverWait(driver, WAIT_TIMEOUT).until(
        EC.invisibility_of_element_located(
            (By.XPATH, "//div[contains(@class,'modal') and .//*[contains(text(),'เพิ่มบัญชีนี้ที่กลุ่ม')]]")
        )
    )
    print("  ✔ เพิ่มบัญชีสำเร็จ!")


# ----------------------------------------------------------
# STEP 2 : ลบบัญชีเก่า
# ----------------------------------------------------------
def remove_old_account(driver: webdriver.Chrome, account_name: str):
    print(f"\n[REMOVE] กำลังลบบัญชี: {account_name}")

    max_pages = 5
    removed = False

    for page in range(max_pages):
        # หาแถว account ที่ต้องการลบ
        rows = driver.find_elements(
            By.XPATH,
            f"//tr[.//a[contains(normalize-space(),'{account_name}')]] | "
            f"//div[contains(@class,'account-row') and contains(normalize-space(),'{account_name}')]"
        )

        if rows:
            # กดปุ่ม "ลบ" ในแถวนั้น
            delete_btn = rows[0].find_element(
                By.XPATH,
                ".//button[normalize-space()='ลบ']"
            )
            delete_btn.click()
            print(f"  ✔ กดปุ่ม 'ลบ' สำหรับบัญชี '{account_name}'")
            time.sleep(1.5)

            # ถ้ามี confirm dialog ให้กด confirm
            try:
                confirm = WebDriverWait(driver, 5).until(
                    EC.element_to_be_clickable(
                        (By.XPATH, "//button[contains(@class,'confirm') or normalize-space()='ตกลง' or normalize-space()='ยืนยัน']")
                    )
                )
                confirm.click()
                print("  ✔ ยืนยันการลบแล้ว")
                time.sleep(1.5)
            except TimeoutException:
                pass  # ไม่มี confirm dialog ก็ผ่าน

            removed = True
            print("  ✔ ลบบัญชีสำเร็จ!")
            break

        # ถ้าไม่เจอ ลองไปหน้าถัดไป
        try:
            next_btn = driver.find_element(
                By.XPATH,
                "//button[@aria-label='Next page'] | //a[contains(@class,'next') and not(@disabled)]"
            )
            next_btn.click()
            time.sleep(2)
        except NoSuchElementException:
            break  # ไม่มีหน้าถัดไปแล้ว

    if not removed:
        print(f"  ⚠ ไม่พบบัญชี '{account_name}' ที่จะลบ")


# ----------------------------------------------------------
# MAIN
# ----------------------------------------------------------
def main():
    driver = init_driver()

    try:
        print("=== LINE OA Auto Account Manager ===")
        print(f"URL: {GROUP_SETTINGS_URL}\n")

        print("  → เปิดหน้า Group Settings...")
        driver.get(GROUP_SETTINGS_URL)
        time.sleep(3)  # รอหน้าโหลด

        # STEP 1: เพิ่มบัญชีใหม่
        add_new_account(driver, NEW_ACCOUNT_NAME)

        # STEP 2: ลบบัญชีเก่า
        remove_old_account(driver, OLD_ACCOUNT_NAME)

        print("\n✅ ทำงานเสร็จสิ้น!")

    except Exception as e:
        print(f"\n❌ เกิดข้อผิดพลาด: {e}")
        driver.save_screenshot("error_screenshot.png")
        print("   บันทึก screenshot ไว้ที่ error_screenshot.png")

    finally:
        input("\nกด Enter เพื่อปิด browser...")
        driver.quit()


if __name__ == "__main__":
    main()
